// Original Parallax C# source. The .cs.txt extension permits source-only packaging;
// Rainmeter's once-per-refresh PowerShell query compiles this with Add-Type.
// Uses Windows DXCore and the existing NVIDIA driver. No files, processes or live settings are changed.
// API contract: https://learn.microsoft.com/en-us/windows/win32/dxcore/dxcore-enum-adapters
// ABI reference: https://github.com/microsoft/DirectX-Headers/blob/main/include/directx/dxcore_interface.h
// Public NVAPI ABI / IDs: https://github.com/NVIDIA/nvapi/blob/main/nvapi.h
// https://github.com/NVIDIA/nvapi/blob/main/nvapi_interface.h
// Base/boost graphics clock semantics: https://docs.nvidia.com/nvapi/group__gpuclock.html
// Private read-only RAM type/maker ABI and enum references (not a bundled dependency):
// https://github.com/falahati/NvAPIWrapper/blob/master/NvAPIWrapper/Native/Delegates/GPU.cs
// https://github.com/falahati/NvAPIWrapper/blob/master/NvAPIWrapper/Native/Helpers/FunctionId.cs
// https://github.com/falahati/NvAPIWrapper/blob/master/NvAPIWrapper/Native/GPU/GPUMemoryType.cs
// https://github.com/falahati/NvAPIWrapper/blob/master/NvAPIWrapper/Native/GPU/GPUMemoryMaker.cs
// Extended type codes: Fastfetch 2.67.0 src/detection/gpu/nvapi.h, original source
// reproduced at https://fossies.org/linux/fastfetch/src/detection/gpu/nvapi.h
// Private entry points are optional and unsupported codes remain unknown.
using System;
using System.Globalization;
using System.Runtime.InteropServices;
using System.Text;

public static class ParallaxGpuInfo
{
    private static readonly Guid FactoryId = new Guid("78ee5945-c36e-4b13-a669-005dd11c0f06");
    private static readonly Guid ListId = new Guid("526c7776-40e9-459b-b711-f32ad76dfc28");
    private static readonly Guid AdapterId = new Guid("f0db4c7f-fe5a-42a2-bd62-f2a6cf6fc83e");
    private static readonly Guid GraphicsAttribute = new Guid("0c9ece4d-2f6e-4f01-8c96-e89e331b47b1");
    private const uint InstanceLuid = 0;
    private const uint DriverDescription = 2;
    private const uint DedicatedAdapterMemory = 7;
    private const uint IsHardware = 11;
    private const uint IsIntegrated = 12;
    private const uint HighPerformance = 2;
    private const int MaximumNvidiaGpus = 64;
    private const int MaximumNvidiaClockDomains = 32;
    private const int NvidiaIncompatibleStructVersion = -9;
    private const uint NvidiaBaseClock = 1;
    private const uint NvidiaBoostClock = 2;
    private const uint LoadLibrarySearchSystem32 = 0x00000800;

    [DllImport("dxcore.dll", ExactSpelling = true, CallingConvention = CallingConvention.StdCall)]
    [DefaultDllImportSearchPaths(DllImportSearchPath.System32)]
    private static extern int DXCoreCreateAdapterFactory(ref Guid interfaceId, out IntPtr factory);

    [DllImport("kernel32.dll", CharSet = CharSet.Unicode, ExactSpelling = true, SetLastError = true)]
    [DefaultDllImportSearchPaths(DllImportSearchPath.System32)]
    private static extern IntPtr LoadLibraryExW(string name, IntPtr file, uint flags);

    [DllImport("kernel32.dll", CharSet = CharSet.Ansi, ExactSpelling = true, SetLastError = true)]
    [DefaultDllImportSearchPaths(DllImportSearchPath.System32)]
    private static extern IntPtr GetProcAddress(IntPtr module, string name);

    [DllImport("kernel32.dll", ExactSpelling = true)]
    [DefaultDllImportSearchPaths(DllImportSearchPath.System32)]
    [return: MarshalAs(UnmanagedType.Bool)]
    private static extern bool FreeLibrary(IntPtr module);

    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    private delegate int CreateList(IntPtr self, uint count, [In] Guid[] attributes, ref Guid interfaceId, out IntPtr list);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    private delegate int GetAdapter(IntPtr self, uint index, ref Guid interfaceId, out IntPtr adapter);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    private delegate uint GetCount(IntPtr self);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    [return: MarshalAs(UnmanagedType.I1)]
    private delegate bool GetFlag(IntPtr self);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    [return: MarshalAs(UnmanagedType.I1)]
    private delegate bool Supports(IntPtr self, uint value);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    private delegate int SortList(IntPtr self, uint count, [In] uint[] preferences);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    private delegate int GetProperty(IntPtr self, uint property, UIntPtr bufferSize, IntPtr buffer);
    [UnmanagedFunctionPointer(CallingConvention.StdCall)]
    private delegate int GetPropertySize(IntPtr self, uint property, out UIntPtr size);
    [UnmanagedFunctionPointer(CallingConvention.Cdecl)]
    private delegate IntPtr NvidiaQueryInterface(uint id);
    [UnmanagedFunctionPointer(CallingConvention.Cdecl)]
    private delegate int NvidiaLifecycle();
    [UnmanagedFunctionPointer(CallingConvention.Cdecl)]
    private delegate int NvidiaEnumerate([Out, MarshalAs(UnmanagedType.LPArray, SizeConst = MaximumNvidiaGpus)] IntPtr[] handles, out uint count);
    [UnmanagedFunctionPointer(CallingConvention.Cdecl)]
    private delegate int NvidiaLogicalInfo(IntPtr logicalGpu, ref NvidiaLogicalData data);
    [UnmanagedFunctionPointer(CallingConvention.Cdecl)]
    private delegate int NvidiaScalar(IntPtr physicalGpu, out uint value);
    [UnmanagedFunctionPointer(CallingConvention.Cdecl)]
    private delegate int NvidiaClockQuery(IntPtr physicalGpu, ref NvidiaClockFrequencies data);

    // NV_LOGICAL_GPU_DATA_V1 uses the SDK's pack(8); pointers follow process
    // bitness. AdapterId points to caller-owned eight-byte Windows LUID storage.
    [StructLayout(LayoutKind.Sequential, Pack = 8)]
    private struct NvidiaLogicalData
    {
        public uint Version;
        public IntPtr AdapterId;
        public uint PhysicalCount;
        [MarshalAs(UnmanagedType.ByValArray, SizeConst = MaximumNvidiaGpus)]
        public IntPtr[] PhysicalHandles;
        [MarshalAs(UnmanagedType.ByValArray, SizeConst = 8)]
        public uint[] Reserved;
    }

    [StructLayout(LayoutKind.Sequential, Pack = 8)]
    private struct NvidiaClockDomain
    {
        public uint Flags; // bIsPresent is bit 0; all other bits are reserved.
        public uint Frequency; // kHz.
    }

    // NV_GPU_CLOCK_FREQUENCIES_V2 / version 3 have identical 264-byte layouts.
    // ClockType is the low four bits; the other 28 input bits must remain zero.
    [StructLayout(LayoutKind.Sequential, Pack = 8)]
    private struct NvidiaClockFrequencies
    {
        public uint Version;
        public uint ClockType;
        [MarshalAs(UnmanagedType.ByValArray, SizeConst = MaximumNvidiaClockDomains)]
        public NvidiaClockDomain[] Domains;
    }

    // One-line success protocol, exactly eight fields:
    // OK|model|memoryBytes|memoryType|memoryVendor|memorySource|baseKHz|boostKHz
    // An unknown individual field is "?". PHYSICAL means driver-reported
    // physical framebuffer; DEDICATED means DXCore dedicated-adapter bytes.
    // Base/boost are independent driver-reported graphics clock specifications,
    // never the current clock, memory clock, or an observed maximum.
    // Other results remain NONE, UNAVAILABLE, or AMBIGUOUS.
    // "Main" means the first discrete GPU in Windows' high-performance order.
    // This list covers D3D12 graphics adapters; it is not a display-ownership query.
    // Memory type/vendor are queried from the selected hardware, never inferred
    // from a model name or a machine-specific profile.
    public static string Read()
    {
        return ReadMetadata(false);
    }

    // The skin also needs the selected adapter's exact identity for its
    // persistent temperature provider. No hardware identity is saved to settings.
    public static string ReadForSkin()
    {
        return ReadMetadata(true);
    }

    private static string ReadMetadata(bool includeIdentity)
    {
        IntPtr factory = IntPtr.Zero;
        IntPtr list = IntPtr.Zero;
        IntPtr selected = IntPtr.Zero;
        try
        {
            Guid factoryId = FactoryId;
            Check(DXCoreCreateAdapterFactory(ref factoryId, out factory));
            Guid listId = ListId;
            Check(Method<CreateList>(factory, 3)(factory, 1, new Guid[] { GraphicsAttribute }, ref listId, out list));

            bool canSort = Method<Supports>(list, 8)(list, HighPerformance);
            if (canSort)
                Check(Method<SortList>(list, 7)(list, 1, new uint[] { HighPerformance }));

            uint count = Method<GetCount>(list, 4)(list);
            if (count > 512) return "UNAVAILABLE";
            int discreteCount = 0;
            for (uint index = 0; index < count; ++index)
            {
                IntPtr adapter = IntPtr.Zero;
                try
                {
                    Guid adapterId = AdapterId;
                    Check(Method<GetAdapter>(list, 3)(list, index, ref adapterId, out adapter));
                    if (!Method<GetFlag>(adapter, 3)(adapter)) return "UNAVAILABLE";
                    if (!ReadBoolean(adapter, IsHardware)) continue;
                    if (ReadBoolean(adapter, IsIntegrated)) continue;
                    ++discreteCount;
                    if (selected == IntPtr.Zero)
                    {
                        Marshal.AddRef(adapter);
                        selected = adapter;
                    }
                }
                finally
                {
                    if (adapter != IntPtr.Zero) Marshal.Release(adapter);
                }
            }
            // Device arrival/removal during enumeration invalidates this snapshot.
            if (Method<GetFlag>(list, 5)(list)) return "UNAVAILABLE";
            if (discreteCount == 0) return "NONE";
            if (!canSort && discreteCount > 1) return "AMBIGUOUS";
            string name = ReadName(selected);
            if (String.IsNullOrEmpty(name)) return "UNAVAILABLE";
            ulong value;
            string memory = TryReadUInt64(selected, DedicatedAdapterMemory, out value) && value > 0
                ? value.ToString(CultureInfo.InvariantCulture) : "?";
            string memoryType = "?", memoryVendor = "?";
            string memorySource = "DEDICATED";
            string baseClock = "?", boostClock = "?";
            string adapterIdentity = "?";
            if (TryReadUInt64(selected, InstanceLuid, out value))
            {
                adapterIdentity = value.ToString("X16", CultureInfo.InvariantCulture);
                ReadNvidiaMetadata(value, ref memory, ref memoryType, ref memoryVendor, ref memorySource, ref baseClock, ref boostClock);
            }
            if (!Method<GetFlag>(selected, 3)(selected) || Method<GetFlag>(list, 5)(list)) return "UNAVAILABLE";
            string metadata = String.Join("|", new string[] { "OK", name, memory, memoryType, memoryVendor, memorySource, baseClock, boostClock });
            return metadata + (includeIdentity ? "\nGPU_ADAPTER|1|" + adapterIdentity : "");
        }
        catch
        {
            // Old Windows, unsupported properties, unavailable drivers and query
            // failures must never silently select an integrated or guessed GPU.
            return "UNAVAILABLE";
        }
        finally
        {
            if (selected != IntPtr.Zero) Marshal.Release(selected);
            if (list != IntPtr.Zero) Marshal.Release(list);
            if (factory != IntPtr.Zero) Marshal.Release(factory);
        }
    }

    private static T Method<T>(IntPtr instance, int slot) where T : class
    {
        if (instance == IntPtr.Zero) throw new InvalidOperationException();
        IntPtr address = Marshal.ReadIntPtr(Marshal.ReadIntPtr(instance), slot * IntPtr.Size);
        return (T)(object)Marshal.GetDelegateForFunctionPointer(address, typeof(T));
    }

    private static void Check(int result)
    {
        if (result < 0) Marshal.ThrowExceptionForHR(result);
    }

    private static bool TryReadUInt64(IntPtr adapter, uint property, out ulong value)
    {
        value = 0;
        IntPtr buffer = IntPtr.Zero;
        try
        {
            if (!Method<Supports>(adapter, 5)(adapter, property)) return false;
            UIntPtr size;
            Check(Method<GetPropertySize>(adapter, 7)(adapter, property, out size));
            if (size.ToUInt64() != 8) return false;
            buffer = Marshal.AllocHGlobal(8);
            Marshal.WriteInt64(buffer, 0);
            Check(Method<GetProperty>(adapter, 6)(adapter, property, size, buffer));
            value = unchecked((ulong)Marshal.ReadInt64(buffer));
            return true;
        }
        catch { return false; }
        finally { if (buffer != IntPtr.Zero) Marshal.FreeHGlobal(buffer); }
    }

    private static T NvidiaFunction<T>(NvidiaQueryInterface query, uint id) where T : class
    {
        IntPtr address = query(id);
        return address == IntPtr.Zero ? null
            : (T)(object)Marshal.GetDelegateForFunctionPointer(address, typeof(T));
    }

    private static bool TryNvidiaScalar(NvidiaQueryInterface query, uint id, IntPtr gpu, out uint value)
    {
        value = 0;
        try
        {
            NvidiaScalar read = NvidiaFunction<NvidiaScalar>(query, id);
            return read != null && read(gpu, out value) == 0;
        }
        catch { return false; }
    }

    private static NvidiaLogicalData NewNvidiaLogicalData(IntPtr luidBuffer)
    {
        return new NvidiaLogicalData {
            Version = (uint)Marshal.SizeOf(typeof(NvidiaLogicalData)) | 0x10000,
            AdapterId = luidBuffer,
            PhysicalHandles = new IntPtr[MaximumNvidiaGpus],
            Reserved = new uint[8]
        };
    }

    private static string ReadNvidiaClock(NvidiaQueryInterface query, IntPtr gpu, uint clockType)
    {
        try
        {
            if (clockType != NvidiaBaseClock && clockType != NvidiaBoostClock) return "?";
            NvidiaClockQuery read = NvidiaFunction<NvidiaClockQuery>(query, 0xdcb616c3);
            if (read == null) return "?";
            foreach (uint version in new uint[] { 3, 2 })
            {
                NvidiaClockFrequencies data = new NvidiaClockFrequencies {
                    Version = (uint)Marshal.SizeOf(typeof(NvidiaClockFrequencies)) | (version << 16),
                    ClockType = clockType,
                    Domains = new NvidiaClockDomain[MaximumNvidiaClockDomains]
                };
                int status = read(gpu, ref data);
                if (status == NvidiaIncompatibleStructVersion) continue;
                if (status != 0 || (data.ClockType & 0xf) != clockType || data.Domains == null ||
                    data.Domains.Length != MaximumNvidiaClockDomains) return "?";
                // NVAPI_GPU_PUBLIC_CLOCK_GRAPHICS is domain 0. Domain 4 is
                // memory and must not be used for either requested GPU clock.
                NvidiaClockDomain graphics = data.Domains[0];
                if ((graphics.Flags & 1) == 0 || graphics.Frequency == 0) return "?";
                return graphics.Frequency.ToString(CultureInfo.InvariantCulture);
            }
            // V1 has no ClockType selector; falling back to it could silently
            // substitute a current frequency for the requested specification.
        }
        catch { /* A missing clock feature must not discard other metadata. */ }
        return "?";
    }

    private static void ReadNvidiaMetadata(ulong selectedLuid, ref string memory, ref string memoryType,
        ref string memoryVendor, ref string memorySource, ref string baseClock, ref string boostClock)
    {
        IntPtr module = IntPtr.Zero;
        IntPtr luidBuffer = IntPtr.Zero;
        NvidiaLifecycle unload = null;
        bool initialized = false;
        try
        {
            // Full system path plus restricted dependency search. Windows' normal
            // WOW64 redirection selects the x86 system DLL for a 32-bit process.
            string dll = IntPtr.Size == 8 ? "nvapi64.dll" : "nvapi.dll";
            string path = System.IO.Path.Combine(Environment.SystemDirectory, dll);
            module = LoadLibraryExW(path, IntPtr.Zero, LoadLibrarySearchSystem32);
            if (module == IntPtr.Zero) return;
            IntPtr entry = GetProcAddress(module, "nvapi_QueryInterface");
            if (entry == IntPtr.Zero) return;
            NvidiaQueryInterface query = (NvidiaQueryInterface)Marshal.GetDelegateForFunctionPointer(entry, typeof(NvidiaQueryInterface));
            NvidiaLifecycle initialize = NvidiaFunction<NvidiaLifecycle>(query, 0x0150e828);
            unload = NvidiaFunction<NvidiaLifecycle>(query, 0xd22bdd7e);
            if (initialize == null || unload == null || initialize() != 0) return;
            initialized = true;
            NvidiaEnumerate enumerate = NvidiaFunction<NvidiaEnumerate>(query, 0x48b3ea59);
            NvidiaLogicalInfo logicalInfo = NvidiaFunction<NvidiaLogicalInfo>(query, 0x842b066e);
            if (enumerate == null || logicalInfo == null) return;

            IntPtr[] logicalGpus = new IntPtr[MaximumNvidiaGpus];
            uint count;
            if (enumerate(logicalGpus, out count) != 0 || count > MaximumNvidiaGpus) return;
            luidBuffer = Marshal.AllocHGlobal(8);
            IntPtr selectedPhysical = IntPtr.Zero;
            IntPtr selectedLogical = IntPtr.Zero;
            for (int index = 0; index < count; ++index)
            {
                if (logicalGpus[index] == IntPtr.Zero) return;
                Marshal.WriteInt64(luidBuffer, 0);
                NvidiaLogicalData data = NewNvidiaLogicalData(luidBuffer);
                if (logicalInfo(logicalGpus[index], ref data) != 0) return;
                if (data.AdapterId != luidBuffer || data.PhysicalCount > MaximumNvidiaGpus) return;
                if (unchecked((ulong)Marshal.ReadInt64(luidBuffer)) != selectedLuid) continue;
                // Linked multi-GPU adapters do not establish one physical card.
                // Never choose the first NVIDIA card or guess from its name.
                if (selectedPhysical != IntPtr.Zero || data.PhysicalCount != 1 || data.PhysicalHandles[0] == IntPtr.Zero) return;
                selectedPhysical = data.PhysicalHandles[0];
                selectedLogical = logicalGpus[index];
            }
            if (selectedPhysical == IntPtr.Zero) return;

            uint value;
            string physicalMemory = "?", type = "?", vendor = "?";
            // Public API returns physical framebuffer size in KiB, excluding
            // dedicated system RAM. Cast before multiplication to avoid overflow.
            if (TryNvidiaScalar(query, 0x46fbeb03, selectedPhysical, out value) && value > 0)
                physicalMemory = ((ulong)value * 1024).ToString(CultureInfo.InvariantCulture);
            if (TryNvidiaScalar(query, 0x57f7caac, selectedPhysical, out value))
                type = NvidiaMemoryType(value);
            if (TryNvidiaScalar(query, 0x42aea16a, selectedPhysical, out value))
                vendor = NvidiaMemoryVendor(value);
            string baseFrequency = ReadNvidiaClock(query, selectedPhysical, NvidiaBaseClock);
            string boostFrequency = ReadNvidiaClock(query, selectedPhysical, NvidiaBoostClock);

            // Recheck the association after the bounded read in case topology
            // changed. A failure preserves only the earlier DXCore fallback.
            Marshal.WriteInt64(luidBuffer, 0);
            NvidiaLogicalData verified = NewNvidiaLogicalData(luidBuffer);
            if (logicalInfo(selectedLogical, ref verified) != 0 || verified.AdapterId != luidBuffer ||
                unchecked((ulong)Marshal.ReadInt64(luidBuffer)) != selectedLuid ||
                verified.PhysicalCount != 1 || verified.PhysicalHandles[0] != selectedPhysical) return;
            if (physicalMemory != "?")
            {
                memory = physicalMemory;
                memorySource = "PHYSICAL";
            }
            memoryType = type;
            memoryVendor = vendor;
            baseClock = baseFrequency;
            boostClock = boostFrequency;
        }
        catch { /* Missing/unsupported NVIDIA driver keeps the known DXCore metadata. */ }
        finally
        {
            if (luidBuffer != IntPtr.Zero) Marshal.FreeHGlobal(luidBuffer);
            try { if (initialized) unload(); }
            catch { /* Cleanup must not discard a completed metadata result. */ }
            finally { if (module != IntPtr.Zero) FreeLibrary(module); }
        }
    }

    private static string NvidiaMemoryType(uint value)
    {
        // Verified enum values, not a GPU model database. Unknown/new codes must
        // remain unknown until their meanings have an independent ABI reference.
        switch (value)
        {
            case 1: return "SDRAM";
            case 2: return "DDR";
            case 3: return "DDR2";
            case 4: return "GDDR2";
            case 5: return "GDDR3";
            case 6: return "GDDR4";
            case 7: return "DDR3";
            case 8: return "GDDR5";
            case 9: return "LPDDR2";
            case 10: return "GDDR5X";
            case 11: return "LPDDR3";
            case 12: return "LPDDR4";
            case 13: return "LPDDR5";
            case 14: return "GDDR6";
            case 15: return "GDDR6X";
            case 16: return "GDDR7";
            default: return "?";
        }
    }

    private static string NvidiaMemoryVendor(uint value)
    {
        switch (value)
        {
            case 1: return "Samsung";
            case 2: return "Qimonda";
            case 3: return "Elpida";
            case 4: return "Etron";
            case 5: return "Nanya";
            case 6: return "SK Hynix";
            case 7: return "Mosel";
            case 8: return "Winbond";
            case 9: return "ESMT";
            case 10: return "Micron";
            default: return "?";
        }
    }

    private static bool ReadBoolean(IntPtr adapter, uint property)
    {
        if (!Method<Supports>(adapter, 5)(adapter, property)) throw new NotSupportedException();
        IntPtr buffer = Marshal.AllocHGlobal(1);
        try
        {
            Marshal.WriteByte(buffer, 0);
            Check(Method<GetProperty>(adapter, 6)(adapter, property, new UIntPtr(1), buffer));
            byte value = Marshal.ReadByte(buffer);
            if (value > 1) throw new InvalidOperationException();
            return value != 0;
        }
        finally { Marshal.FreeHGlobal(buffer); }
    }

    private static string ReadName(IntPtr adapter)
    {
        if (!Method<Supports>(adapter, 5)(adapter, DriverDescription)) throw new NotSupportedException();
        UIntPtr nativeSize;
        Check(Method<GetPropertySize>(adapter, 7)(adapter, DriverDescription, out nativeSize));
        ulong length = nativeSize.ToUInt64();
        if (length == 0 || length > 16384) throw new InvalidOperationException();
        IntPtr buffer = Marshal.AllocHGlobal((int)length);
        try
        {
            Check(Method<GetProperty>(adapter, 6)(adapter, DriverDescription, nativeSize, buffer));
            byte[] bytes = new byte[(int)length];
            Marshal.Copy(buffer, bytes, 0, bytes.Length);
            int end = Array.IndexOf(bytes, (byte)0);
            if (end < 0) throw new InvalidOperationException();
            string name = new UTF8Encoding(false, true).GetString(bytes, 0, end);
            StringBuilder clean = new StringBuilder();
            bool space = false;
            foreach (char character in name)
            {
                if (character == '|' || Char.IsControl(character) || Char.IsWhiteSpace(character))
                {
                    space = clean.Length > 0;
                    continue;
                }
                if (space) clean.Append(' ');
                clean.Append(character);
                space = false;
            }
            return clean.ToString();
        }
        finally { Marshal.FreeHGlobal(buffer); }
    }
}
